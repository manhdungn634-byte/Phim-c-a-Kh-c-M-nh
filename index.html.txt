  <script>
    // ============================================================
    // Anime Xem Liền — Gọi Jikan API v4
    // Docs: https://docs.api.jikan.moe/
    // Giới hạn: 3 req/giây, 60 req/phút
    // ============================================================

    const BASE_URL = 'https://api.jikan.moe/v4';
    const CACHE_TTL = 5 * 60 * 1000; // 5 phút

    // ---------- Cache đơn giản ----------
    const cache = new Map();
    function getCache(key) {
      const entry = cache.get(key);
      if (!entry) return null;
      if (Date.now() - entry.time > CACHE_TTL) {
        cache.delete(key);
        return null;
      }
      return entry.data;
    }
    function setCache(key, data) {
      cache.set(key, { data, time: Date.now() });
    }

    // ---------- DOM refs ----------
    const grid = document.getElementById('animeGrid');
    const statusEl = document.getElementById('status');
    const searchInput = document.getElementById('searchInput');
    const searchBtn = document.getElementById('searchBtn');
    const topBtn = document.getElementById('topBtn');
    const logo = document.getElementById('logo');
    const modalOverlay = document.getElementById('modalOverlay');
    const modalContent = document.getElementById('modalContent');
    const modalClose = document.getElementById('modalClose');

    // ---------- State ----------
    let searchTimeout = null;
    let lastQuery = '';
    let currentRequest = 0;

    // ---------- Gọi API ----------
    async function fetchAPI(endpoint) {
      const cached = getCache(endpoint);
      if (cached) return cached;

      const res = await fetch(`${BASE_URL}${endpoint}`);

      if (res.status === 429) {
        throw new Error('Đã đạt giới hạn request. Vui lòng đợi vài giây.');
      }
      if (res.status === 404) {
        throw new Error('Không tìm thấy dữ liệu.');
      }
      if (!res.ok) {
        throw new Error(`Lỗi API (${res.status}). Vui lòng thử lại.`);
      }

      const data = await res.json();
      setCache(endpoint, data);
      return data;
    }

    // ---------- Render ----------
    function renderGrid(animeList) {
      if (!animeList || animeList.length === 0) {
        grid.innerHTML = '';
        setStatus('Không tìm thấy anime nào.', 'error');
        return;
      }

      setStatus('');

      grid.innerHTML = animeList
        .map((anime) => {
          const img =
            anime.images?.jpg?.image_url ||
            anime.images?.webp?.image_url ||
            'https://via.placeholder.com/225x318/2a2a30/666?text=No+Image';
          const title = anime.title_english || anime.title || 'Không rõ tên';
          const score = anime.score ? anime.score.toFixed(1) : '—';
          const type = anime.type || '?';

          return `
            <article class="card" data-id="${anime.mal_id}" tabindex="0" role="button" aria-label="Xem chi tiết ${escapeHtml(title)}">
              <img src="${escapeHtml(img)}" alt="${escapeHtml(title)}" loading="lazy" />
              <div class="card-info">
                <h3 title="${escapeHtml(title)}">${escapeHtml(title)}</h3>
                <div class="meta">
                  <span>${escapeHtml(type)}</span>
                  <span class="score">★ ${score}</span>
                </div>
              </div>
            </article>
          `;
        })
        .join('');

      grid.querySelectorAll('.card').forEach((card) => {
        card.addEventListener('click', () => showDetail(card.dataset.id));
        card.addEventListener('keydown', (e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault();
            showDetail(card.dataset.id);
          }
        });
      });
    }

    function setStatus(msg, type = '') {
      statusEl.textContent = msg;
      statusEl.className = 'status' + (type ? ' ' + type : '');
    }

    // ---------- Top anime ----------
    async function loadTopAnime() {
      const reqId = ++currentRequest;
      setStatus('Đang tải top anime...');
      grid.innerHTML = '';

      try {
        const data = await fetchAPI('/top/anime?limit=24');
        if (reqId !== currentRequest) return;
        renderGrid(data.data);
      } catch (err) {
        if (reqId !== currentRequest) return;
        setStatus(err.message, 'error');
      }
    }

    // ---------- Tìm kiếm ----------
    async function searchAnime(query) {
      const q = query.trim();
      if (!q) {
        loadTopAnime();
        return;
      }

      const reqId = ++currentRequest;
      setStatus(`Đang tìm "${q}"...`);
      grid.innerHTML = '';

      try {
        const data = await fetchAPI(
          `/anime?q=${encodeURIComponent(q)}&limit=24&sfw=true&order_by=score&sort=desc`
        );
        if (reqId !== currentRequest) return;
        renderGrid(data.data);
      } catch (err) {
        if (reqId !== currentRequest) return;
        setStatus(err.message, 'error');
      }
    }

    // ---------- Chi tiết ----------
    async function showDetail(malId) {
      openModal();
      modalContent.innerHTML = '<p style="color:#999;">Đang tải chi tiết...</p>';

      try {
        const data = await fetchAPI(`/anime/${malId}/full`);
        const a = data.data;

        const img =
          a.images?.jpg?.large_image_url ||
          a.images?.jpg?.image_url ||
          '';
        const title = a.title_english || a.title || 'Không rõ';
        const genres = (a.genres || [])
          .map((g) => `<span class="tag">${escapeHtml(g.name)}</span>`)
          .join('');
        const studios = (a.studios || [])
          .map((s) => `<span class="tag">🏢 ${escapeHtml(s.name)}</span>`)
          .join('');
        const synopsis = a.synopsis || 'Chưa có mô tả cho anime này.';

        modalContent.innerHTML = `
          <img src="${escapeHtml(img)}" alt="${escapeHtml(title)}" />
          <div class="modal-details">
            <h2>${escapeHtml(title)}</h2>
            <div class="stat"><strong>Điểm:</strong> ${a.score ?? '—'} / 10${
          a.scored_by ? ` (${a.scored_by.toLocaleString('vi-VN')} lượt)` : ''
        }</div>
            <div class="stat"><strong>Loại:</strong> ${escapeHtml(a.type || '?')} · <strong>Số tập:</strong> ${a.episodes ?? '?'}</div>
            <div class="stat"><strong>Trạng thái:</strong> ${escapeHtml(a.status || '?')}</div>
            ${a.rank ? `<div class="stat"><strong>Xếp hạng:</strong> #${a.rank}</div>` : ''}
            <div class="tags">${genres}${studios}</div>
            <p>${escapeHtml(synopsis)}</p>
          </div>
        `;
      } catch (err) {
        modalContent.innerHTML = `<p style="color:#ff6b6b;">${escapeHtml(err.message)}</p>`;
      }
    }

    // ---------- Modal ----------
    function openModal() {
      modalOverlay.classList.add('active');
      modalOverlay.setAttribute('aria-hidden', 'false');
      document.body.style.overflow = 'hidden';
    }
    function closeModal() {
      modalOverlay.classList.remove('active');
      modalOverlay.setAttribute('aria-hidden', 'true');
      document.body.style.overflow = '';
    }

    modalClose.addEventListener('click', closeModal);
    modalOverlay.addEventListener('click', (e) => {
      if (e.target === modalOverlay) closeModal();
    });
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && modalOverlay.classList.contains('active')) {
        closeModal();
      }
    });

    // ---------- Sự kiện UI ----------
    searchInput.addEventListener('input', (e) => {
      clearTimeout(searchTimeout);
      const q = e.target.value;
      searchTimeout = setTimeout(() => {
        if (q.trim() !== lastQuery) {
          lastQuery = q.trim();
          searchAnime(q);
        }
      }, 500);
    });

    searchBtn.addEventListener('click', () => {
      clearTimeout(searchTimeout);
      lastQuery = searchInput.value.trim();
      searchAnime(searchInput.value);
    });

    searchInput.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        clearTimeout(searchTimeout);
        lastQuery = searchInput.value.trim();
        searchAnime(searchInput.value);
      }
    });

    topBtn.addEventListener('click', () => {
      searchInput.value = '';
      lastQuery = '';
      loadTopAnime();
    });

    logo.addEventListener('click', (e) => {
      e.preventDefault();
      searchInput.value = '';
      lastQuery = '';
      loadTopAnime();
    });

    // ---------- Tiện ích ----------
    function escapeHtml(str) {
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    // ---------- Khởi chạy ----------
    loadTopAnime();
  </script>
